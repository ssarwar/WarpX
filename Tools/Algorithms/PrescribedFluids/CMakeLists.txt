cmake_minimum_required(VERSION 3.25)
project(PrescribedFluids LANGUAGES C CXX)
find_package(AMReX REQUIRED CONFIG)
if(AMReX_CUDA_FOUND)
    enable_language(CUDA)
    set_source_files_properties(test_profiles.cpp test_poynting.cpp PROPERTIES LANGUAGE CUDA)
elseif(AMReX_HIP_FOUND)
    enable_language(HIP)
    set_source_files_properties(test_profiles.cpp test_poynting.cpp PROPERTIES LANGUAGE HIP)
endif()
add_executable(test_fluid_profiles test_profiles.cpp)
target_compile_features(test_fluid_profiles PRIVATE cxx_std_17)
target_include_directories(test_fluid_profiles PRIVATE ../../../Source)
target_link_libraries(test_fluid_profiles PRIVATE AMReX::amrex)
enable_testing()
add_test(NAME fluid_profiles COMMAND test_fluid_profiles)
set_tests_properties(fluid_profiles PROPERTIES TIMEOUT 30)

# These tests exercise dimension-specific interpolation helpers on explicit
# Array4 storage; they do not instantiate a simulation of a different geometry.
foreach(geometry IN ITEMS 1D_Z RZ 3D)
    add_executable(test_poynting_${geometry} test_poynting.cpp)
    target_compile_features(test_poynting_${geometry} PRIVATE cxx_std_17)
    target_compile_definitions(test_poynting_${geometry} PRIVATE WARPX_DIM_${geometry})
    target_include_directories(test_poynting_${geometry} PRIVATE ../../../Source)
    target_link_libraries(test_poynting_${geometry} PRIVATE AMReX::amrex)
    add_test(NAME poynting_${geometry} COMMAND test_poynting_${geometry})
    set_tests_properties(poynting_${geometry} PROPERTIES TIMEOUT 30)
endforeach()
